PCA — 2D

{
  const d = data.pca_2d;
  const colorField = data.color_fields[0];
  const groups = d3.group(d.points, p => p[colorField] ?? "Unknown");
  const traces = Array.from(groups, ([name, pts]) => ({
    x: pts.map(p => p.PC1),
    y: pts.map(p => p.PC2),
    text: pts.map(p => p.Model),
    mode: "markers",
    type: "scatter",
    name: String(name),
    marker: {size: 8, opacity: 0.85},
    hovertemplate: "%{text}<br>PC1: %{x:.3f}<br>PC2: %{y:.3f}<extra>%{fullData.name}</extra>",
  }));

  const div = document.createElement("div");
  Plotly.newPlot(div, traces, {
    title: `PCA 2D — ${data.name}<br><sub>PC1 ${(d.variance[0]*100).toFixed(1)}% var · PC2 ${(d.variance[1]*100).toFixed(1)}% var</sub>`,
    xaxis: {title: `PC1 (${(d.variance[0]*100).toFixed(1)}%)`},
    yaxis: {title: `PC2 (${(d.variance[1]*100).toFixed(1)}%)`},
    width: 1280, height: 720,
    paper_bgcolor: "white", plot_bgcolor: "white",
  }, {responsive: true});
  return div;
}

Custom-Axis Scatter

viewof scatter_x = Inputs.select(data.benchmarks, {
  label: "X Axis",
  value: data.benchmarks[0],
})
viewof scatter_y = Inputs.select(data.benchmarks, {
  label: "Y Axis",
  value: data.benchmarks.length > 1 ? data.benchmarks[1] : data.benchmarks[0],
})
viewof scatter_color = Inputs.select(data.color_fields, {
  label: "Color By",
  value: data.color_fields[0],
})
{
  const pts = data.scatter_data.points.filter(
    p => p[scatter_x] != null && p[scatter_y] != null
  );
  const groups = d3.group(pts, p => String(p[scatter_color] ?? "Unknown"));
  const traces = Array.from(groups, ([name, points]) => ({
    x: points.map(p => p[scatter_x]),
    y: points.map(p => p[scatter_y]),
    text: points.map(p => p.Model),
    mode: "markers",
    type: "scatter",
    name,
    marker: {size: 9, opacity: 0.8},
    hovertemplate:
      `%{text}<br>${scatter_x}: %{x:.3f}<br>${scatter_y}: %{y:.3f}<extra>%{fullData.name}</extra>`,
  }));

  const n = pts.length;
  const div = document.createElement("div");
  Plotly.newPlot(div, traces, {
    title: `${scatter_y} vs ${scatter_x}<br><sub>${n} models</sub>`,
    xaxis: {title: scatter_x},
    yaxis: {title: scatter_y},
    width: 1280, height: 720,
    paper_bgcolor: "white", plot_bgcolor: "white",
  }, {responsive: true});
  return div;
}

Classifier

viewof clf_type = Inputs.select(
  new Map([["SVM (RBF)", "svm"], ["Decision Tree", "dt"], ["Logistic Regression", "lr"]]),
  {label: "Classifier", value: "svm"}
)
viewof clf_target = Inputs.select(
  new Map(Object.entries(data.classifier_targets).map(([k, v]) => [v.label, k])),
  {label: "Target"}
)
viewof clf_origins = Inputs.select(
  new Map(Object.entries(data.classifier_origin_filters).map(([k, v]) => [v.label, k])),
  {label: "Origin Filter"}
)
clf_key = `${clf_type}|${clf_target}|${clf_origins}`
clf_result = data.classifier_results[clf_key] ?? null

Confusion Matrix

{
  if (!clf_result)
    return html`<p style="color:#888"><em>Not enough data for this combination.</em></p>`;

  const d = clf_result;
  const cm_norm = d.cm.map(row => {
    const s = row.reduce((a, b) => a + b, 0) || 1;
    return row.map(v => v / s);
  });

  const traces = [{
    z: cm_norm,
    x: d.labels,
    y: d.labels,
    type: "heatmap",
    colorscale: "Blues",
    zmin: 0, zmax: 1,
    text: d.cm,
    texttemplate: "%{text}",
    hovertemplate: "True: %{y}<br>Pred: %{x}<br>Count: %{text}<br>Rate: %{z:.1%}<extra></extra>",
  }];

  const div = document.createElement("div");
  Plotly.newPlot(div, traces, {
    title: `Confusion Matrix — ${d.clf_name} predicting ${d.target_label}<br><sub>Accuracy: ${(d.accuracy*100).toFixed(1)}% · ${d.n_test} test samples · Origins: ${d.origin_label}</sub>`,
    xaxis: {title: "Predicted"},
    yaxis: {title: "True", autorange: "reversed"},
    width: 900, height: 720,
    paper_bgcolor: "white", plot_bgcolor: "white",
  }, {responsive: true});
  return div;
}

ROC Curve

{
  if (!clf_result) return html``;

  const d = clf_result;
  const traces = Object.values(d.roc).map(r => ({
    x: r.fpr,
    y: r.tpr,
    mode: "lines",
    name: `${r.label} (AUC=${r.auc.toFixed(3)})`,
  }));

  traces.push({
    x: [0, 1], y: [0, 1],
    mode: "lines",
    line: {dash: "dash", color: "gray"},
    name: "Random (AUC=0.500)",
  });

  const div = document.createElement("div");
  Plotly.newPlot(div, traces, {
    title: `ROC Curve — ${d.clf_name} predicting ${d.target_label}<br><sub>Macro AUC: ${d.macro_auc.toFixed(3)}</sub>`,
    xaxis: {title: "False Positive Rate", constrain: "domain", scaleanchor: "y"},
    yaxis: {title: "True Positive Rate", constrain: "domain"},
    width: 800, height: 800,
    paper_bgcolor: "white", plot_bgcolor: "white",
    legend: {x: 0.6, y: 0.05},
  }, {responsive: true});
  return div;
}

Feature Importances

{
  if (!clf_result || !clf_result.importances)
    return html`<p style="color:#888"><em>No feature importances for this classifier (SVM has none natively).</em></p>`;

  const d = clf_result;
  const entries = Object.entries(d.importances).sort((a, b) => a[1] - b[1]);

  const traces = [{
    y: entries.map(e => e[0]),
    x: entries.map(e => e[1]),
    type: "bar",
    orientation: "h",
  }];

  const div = document.createElement("div");
  Plotly.newPlot(div, traces, {
    title: `Feature Importances — ${d.clf_name}`,
    xaxis: {title: "Importance"},
    width: 1280, height: 720,
    paper_bgcolor: "white", plot_bgcolor: "white",
  }, {responsive: true});
  return div;
}
